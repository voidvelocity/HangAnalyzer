cmake_minimum_required(VERSION 3.16)
project(ascend_hang_flight LANGUAGES CXX)
add_library(flightrecorder SHARED src/flight_recorder.cpp src/flight_checkpoint.cpp)
target_include_directories(flightrecorder PUBLIC include)
target_compile_features(flightrecorder PUBLIC cxx_std_17)
target_compile_options(flightrecorder PRIVATE -O2 -Wall -Wextra)

set(CANN_HOME "" CACHE PATH "CANN toolkit root (for optional checkpoint adapter)")
if(CANN_HOME)
  find_path(CANN_INCLUDE_DIR acl/acl_rt.h
            PATHS "${CANN_HOME}/include" "${CANN_HOME}/aarch64-linux/include" NO_DEFAULT_PATH)
  find_library(CANN_ASCENDCL ascendcl
               PATHS "${CANN_HOME}/lib64" "${CANN_HOME}/aarch64-linux/lib64" NO_DEFAULT_PATH)
  if(NOT CANN_INCLUDE_DIR OR NOT CANN_ASCENDCL)
    message(FATAL_ERROR "CANN_HOME does not contain acl/acl_rt.h and libascendcl")
  endif()
  add_library(flightcheckpoint_cann SHARED src/cann_checkpoint.cpp)
  target_include_directories(flightcheckpoint_cann PRIVATE include "${CANN_INCLUDE_DIR}")
  target_link_libraries(flightcheckpoint_cann PRIVATE flightrecorder "${CANN_ASCENDCL}")
  target_compile_features(flightcheckpoint_cann PRIVATE cxx_std_17)
  set_target_properties(flightcheckpoint_cann PROPERTIES BUILD_RPATH "$ORIGIN;${CANN_HOME}/lib64;${CANN_HOME}/aarch64-linux/lib64")
  find_path(MSPTI_INCLUDE_DIR mspti.h PATHS "${CANN_HOME}/tools/mspti/include" NO_DEFAULT_PATH)
  find_library(MSPTI_LIBRARY mspti PATHS "${CANN_HOME}/tools/mspti/lib64" NO_DEFAULT_PATH)
  if(MSPTI_INCLUDE_DIR AND MSPTI_LIBRARY)
    add_library(hangmspti SHARED src/hang_mspti.cpp)
    target_include_directories(hangmspti PRIVATE include "${MSPTI_INCLUDE_DIR}")
    target_link_libraries(hangmspti PRIVATE "${MSPTI_LIBRARY}")
    target_compile_features(hangmspti PRIVATE cxx_std_17)
    target_compile_options(hangmspti PRIVATE -O2 -Wall -Wextra)
    set_target_properties(hangmspti PROPERTIES BUILD_RPATH "${CANN_HOME}/tools/mspti/lib64")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(recorder_stress tests/recorder_stress.cpp)
  target_link_libraries(recorder_stress PRIVATE flightrecorder Threads::Threads)
  add_test(NAME recorder_stress COMMAND recorder_stress)
  add_executable(checkpoint_pool_test tests/checkpoint_pool_test.cpp)
  target_link_libraries(checkpoint_pool_test PRIVATE flightrecorder Threads::Threads)
  add_test(NAME checkpoint_pool_test COMMAND checkpoint_pool_test)
endif()
